let diveCount = 0;

function getTargetRow(depth, gas) {
  for (let row of TABLE1) {
    let matchDepth = row.depth;
    if (gas === 'air') matchDepth = row.air;
    if (gas === 'ean32') matchDepth = row.ean32;
    if (gas === 'ean36') matchDepth = row.ean36;
    if (matchDepth >= depth) return row;
  }
  return null;
}

function calculateAllDives() {
  let currentGroup = '';

  for (let i = 1; i <= diveCount; i++) {
    const gas = document.getElementById(`gas_${i}`).value;
    const depth = parseFloat(document.getElementById(`depth_${i}`).value);
    const time = parseFloat(document.getElementById(`time_${i}`).value);
    const resultDiv = document.getElementById(`result_${i}`);

    if (!depth || !time) {
      resultDiv.style.display = 'block';
      resultDiv.innerHTML = `<span class="warning">⚠️ 請輸入深度與底時間！</span>`;
      break;
    }

    let rnt = 0;
    let sitMins = 0;
    let groupBeforeDive = '';

    if (i > 1) {
      const sitH = parseInt(document.getElementById(`sitH_${i}`).value) || 0;
      const sitM = parseInt(document.getElementById(`sitM_${i}`).value) || 0;
      sitMins = (sitH * 60) + sitM;

      let newGroup = 'A';
      const ranges = TABLE2_RANGES[currentGroup];
      if (ranges) {
        for (let r of ranges) {
          if (sitMins >= r[0] && sitMins <= r[1]) {
            newGroup = r[2];
            break;
          }
          if (sitMins > ranges[ranges.length - 1][1]) newGroup = 'A';
        }
      }
      groupBeforeDive = newGroup;

      const row = getTargetRow(depth, gas);
      if (row) {
        const depthKey = row.air;
        const rntMap = TABLE3_RNT[depthKey] || {};
        rnt = rntMap[groupBeforeDive] || 0;
      }
    }

    const row = getTargetRow(depth, gas);
    if (!row) {
      resultDiv.style.display = 'block';
      resultDiv.innerHTML = `<span class="warning">⚠️ 超出安全深度極限！</span>`;
      break;
    }

    const maxAllowedBT = row.ndl - rnt;
    const totalEquivTime = time + rnt;

    let warningText = '';
    if (time > maxAllowedBT) {
      warningText = `<br><span class="warning">❌ 超出免減壓極限！最大建議底時間為 <b>${maxAllowedBT} mins</b></span>`;
    }

    let exitGroup = 'A';
    for (let k = 0; k < row.limits.length; k++) {
      if (totalEquivTime <= row.limits[k]) {
        exitGroup = GROUPS[k];
        break;
      }
      if (k === row.limits.length - 1) exitGroup = GROUPS[k+1];
    }

    currentGroup = exitGroup;

    resultDiv.style.display = 'block';
    let sitInfo = i > 1 ? `<div class="result-row"><span>SIT 時間:</span> <span>${sitMins} 分鐘 (進入 ${groupBeforeDive} 組)</span></div>` : '';
    let rntInfo = i > 1 ? `<div class="result-row"><span>餘氮時間 (RNT):</span> <span class="highlight">${rnt} mins</span></div>` : '';

    resultDiv.innerHTML = `
      ${sitInfo}
      ${rntInfo}
      <div class="result-row"><span>最大允許底時間 (Adjusted NDL):</span> <span class="highlight">${maxAllowedBT} mins</span></div>
      <div class="result-row"><span>出水壓力群組:</span> <span class="badge">${exitGroup}</span></div>
      ${warningText}
    `;
  }
}
